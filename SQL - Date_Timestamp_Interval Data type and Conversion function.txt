------------------------CONVERSION FUNCTION

--to_char(): Converts a date or number into a string.
--to_date(): Converts a string into a date data type.
--to_number(): Converts a value to a number data type.
--to_timestamp(): Converts a string into a timestamp data type.

------------------------DATA TYPES

--Interval: Stores a period of time (e.g., years, days, hours).
--Timestamp: Stores a specific date and time, including fractional seconds.


CREATE TABLE EMP2(
    NAME VARCHAR(30),
    DOB DATE,
    JOINING_DTE TIMESTAMP,
    EXPERIENCE INTERVAL YEAR TO MONTH,
    WORKING_HRS INTERVAL DAY TO SECOND
    );

SELECT NAME, DOB, JOINING_DTE, EXPERIENCE, WORKING_HRS,SALARY, COUNT(*) FROM EMP2
GROUP BY NAME, DOB, JOINING_DTE, EXPERIENCE, WORKING_HRS,SALARY

---------------------- DELETE DUPLICATE RECORDS
DELETE FROM EMP2 WHERE ROWID NOT IN(
SELECT MAX(ROWID) FROM EMP2
GROUP BY NAME, DOB, JOINING_DTE, EXPERIENCE, WORKING_HRS,SALARY);

------OR

----------------- FIND MIX, MAX, AVG OF SALARY
SELECT MIN(SALARY), MAX(SALARY), AVG(SALARY) FROM EMPLOYEES;

----------------- FIND MIX, MAX, AVG OF SALARY for each department.
SELECT DEPARTMENT_ID, 
    MIN(SALARY), 
    MAX(SALARY), 
    AVG(SALARY) 
FROM EMPLOYEES
    GROUP BY DEPARTMENT_ID
    ORDER BY DEPARTMENT_ID;
