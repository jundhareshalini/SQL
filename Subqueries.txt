---RETRIEVE THE COUNTRY DETAILS FOR REGION EUROPE AND AMERICAS

SELECT * FROM COUNTRIES WHERE REGION_ID IN (
    SELECT REGION_ID FROM REGIONS WHERE REGION_NAME IN ('Europe','Americas')
);

 -------------------- ALL
 
SELECT * FROM COUNTRIES WHERE REGION_ID = ALL (
    SELECT REGION_ID FROM REGIONS WHERE REGION_NAME IN ('Europe','Americas')
);

 -------------------- ANY
 
SELECT * FROM COUNTRIES WHERE REGION_ID = ANY (
    SELECT REGION_ID FROM REGIONS WHERE REGION_NAME IN ('Europe','Americas')
);
