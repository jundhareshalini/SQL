--Syntax: LAG(column, offset, default_value) OVER (ORDER BY ordering_column)

--The LAG function allows you to access the value of a given column from the previous row in the result set.

--The LEAD function, on the other hand, allows you to access the value of a given column from the next row in the result set

------ CASE STATEMENT

SELECT PRODUCT_ID,
    SALE_DATE,
    LAG(REVENUE) OVER(PARTITION BY PRODUCT_ID ORDER BY SALE_DATE) AS PREV_REV,
    REVENUE AS CURRENT_REVENUE,
    LEAD(REVENUE) OVER(PARTITION BY PRODUCT_ID ORDER BY SALE_DATE) AS NEXT_REV,
    CASE
        WHEN REVENUE > LAG(REVENUE) OVER(PARTITION BY PRODUCT_ID ORDER BY SALE_DATE)
        AND REVENUE > LEAD(REVENUE) OVER(PARTITION BY PRODUCT_ID ORDER BY SALE_DATE) THEN 'HIGH'
    
        WHEN REVENUE < LAG(REVENUE) OVER(PARTITION BY PRODUCT_ID ORDER BY SALE_DATE)
        AND REVENUE < LEAD(REVENUE) OVER(PARTITION BY PRODUCT_ID ORDER BY SALE_DATE) THEN 'LOW'
        
        ELSE 'AVERAGE'
    END AS SALES_STATISTICS
FROM SALES2;
