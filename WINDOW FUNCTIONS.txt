-----RANK, DENSE_RANK, ROW_NUMBER

SELECT EMPLOYEE_ID,
    FIRST_NAME, 
    LAST_NAME,
    SALARY,
    RANK() OVER(ORDER BY SALARY) RNK,
    DENSE_RANK() OVER(ORDER BY SALARY) DRNK,
    ROW_NUMBER() OVER(ORDER BY SALARY) RNUM
FROM EMPLOYEES;

-------RANK()

SELECT EMPLOYEE_ID,
    FIRST_NAME, 
    LAST_NAME,
    SALARY,
    RANK() OVER(ORDER BY SALARY)
FROM EMPLOYEES;

----- find employees who have the 4th highest salary rank using the RANK()

SELECT * FROM(
SELECT EMPLOYEE_ID,
    FIRST_NAME, 
    LAST_NAME,
    SALARY,
    RANK() OVER(ORDER BY SALARY DESC) RNK
FROM EMPLOYEES)
WHERE RNK = 4;


-------DENSE RANK()

SELECT EMPLOYEE_ID,
    FIRST_NAME, 
    LAST_NAME,
    SALARY,
    DENSE_RANK() OVER(ORDER BY SALARY) DRANK
FROM EMPLOYEES;

---finds employees who earn the 4th highest distinct salary

SELECT FIRST_NAME, LAST_NAME, DRANK FROM(
SELECT EMPLOYEE_ID,
    FIRST_NAME, 
    LAST_NAME,
    SALARY,
    DENSE_RANK() OVER(ORDER BY SALARY DESC) DRANK
FROM EMPLOYEES)
WHERE DRANK = 4;

-------ROW NUMBER()

SELECT EMPLOYEE_ID,
    FIRST_NAME, 
    LAST_NAME,
    SALARY,
    ROW_NUMBER() OVER(ORDER BY SALARY)
FROM EMPLOYEES;

SELECT * FROM (
SELECT EMPLOYEE_ID,
    FIRST_NAME, 
    LAST_NAME,
    SALARY,
    ROW_NUMBER() OVER(ORDER BY SALARY) RNUM
FROM EMPLOYEES) TMP
WHERE RNUM IN (25,26);



-------------- FIND THE HIGHEST AND LOWEST SALARY FROM TABLE IN ONE QUERY

SELECT * FROM (
    SELECT EMPLOYEE_ID,
        FIRST_NAME,
        LAST_NAME,
        SALARY,
        ROW_NUMBER() OVER (ORDER BY SALARY) RNUM
    FROM EMPLOYEES)
WHERE RNUM = 1
UNION
SELECT * FROM (
    SELECT EMPLOYEE_ID,
        FIRST_NAME,
        LAST_NAME,
        SALARY,
        ROW_NUMBER() OVER (ORDER BY SALARY DESC) RNUM
    FROM EMPLOYEES)
WHERE RNUM = 1;


-------------- FIND THE HIGHEST AND LOWEST SALARY FROM TABLE IN ONE QUERY

WITH CTE1 AS
(SELECT * FROM EMPLOYEES ORDER BY SALARY FETCH FIRST ROW ONLY),
CTE2 AS (SELECT * FROM EMPLOYEES ORDER BY SALARY DESC FETCH FIRST ROW ONLY)
SELECT * FROM CTE1 UNION SELECT * FROM CTE2;


-------------- PARTITION BY

-----FIND HIGHEST SALARY OF EACH DEPARTMENT

SELECT * FROM(
SELECT EMPLOYEE_ID,
    FIRST_NAME, 
    LAST_NAME,
    SALARY,
    DEPARTMENT_ID,
    DENSE_RANK() OVER(PARTITION BY DEPARTMENT_ID ORDER BY SALARY DESC) DRANK
FROM EMPLOYEES)
WHERE DRANK = 1;
