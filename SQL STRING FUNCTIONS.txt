
------FIND DUPLICATE REOCRDS

SELECT EMP_ID, EMP_NAME, SALARY, COUNT(*) FROM EMP
GROUP BY EMP_ID, EMP_NAME, SALARY
HAVING COUNT(*) > 1;

-----OR

WITH CTE AS(
SELECT EMP_ID, 
    EMP_NAME, 
    SALARY, 
    ROW_NUMBER() OVER(PARTITION BY EMP_ID, EMP_NAME, SALARY ORDER BY EMP_ID) AS RNUM
FROM EMP
)
SELECT * FROM CTE WHERE RNUM > 1;

---------------- WRITE STRING @GMAIL.COM FROM EMAIL

SELECT NAME,
    SUBSTR(EMAIL,(INSTR(EMAIL,'@',1,1))) AS SUBSTRING
    FROM EMAIL;


---------------- WRITE STRING GMAIL.COM FROM EMAIL

SELECT NAME,
    SUBSTR(EMAIL,(INSTR(EMAIL,'@',1,1)+1)) AS SUBSTRING
    FROM EMAIL;


---------------- WRITE STRING FROM STARTING TO BEFORE @ FROM EMAIL

SELECT NAME,
    SUBSTR(EMAIL,1,INSTR(EMAIL,'@',1,1)-1) AS SUBSTRING
    FROM EMAIL;
